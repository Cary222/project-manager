# AI 思考流程重构方案（终版 v4）

## 核心变化：删除 ExecutionBus

| v3 | v4 |
|----|-----|
| Node → ExecutionBus → Route → SSE | graph.stream() → Route → TimelineAdapter → SSE |
| 每会话 ExecutionRuntime | 无需 Runtime，直接用 Route 边界 |
| Event 需要手动 publish | Chunk 已有，自动流转 |

**理由**：graph.stream() 本身就是事件流，ExecutionBus 是重复包装。

---

## 最终架构

```
LangGraph
      │
graph.stream()
      │
──── Route ───────────────────────────────────────────────
      │                                                     │
TimelineAdapter.fromGraph(chunk)                            │
（GraphChunk → TimelineCommand）                            │
      │                                                     │
TimelineStore (扁平 Map)                                    │
      │                                                     │
TreeBuilder                                                │
      │                                                     │
React Timeline UI                                          │
────────────────────────────────────────────────────────────
      │
以后 Workflow.stream() → TimelineAdapter.fromWorkflow()
以后 MCP.stream()     → TimelineAdapter.fromMCP()
```

## 3. TimelineStore（扁平 Map）

`features/ai/lib/timeline-store.ts`

```typescript
import type { TaskRecord } from "@/features/ai/types/timeline";

type UpdateCallback = (tasks: Map<string, TaskRecord>) => void;

export class TimelineStore {
  private tasks: Map<string, TaskRecord> = new Map();
  private listeners: Set<UpdateCallback> = new Set();

  /**
   * 应用 TimelineCommand
   */
  applyCommand(cmd: TimelineCommand): void {
    switch (cmd.op) {
      case "create":
        this.tasks.set(cmd.task.id, cmd.task);
        break;
      case "update":
        const existing = this.tasks.get(cmd.id);
        if (existing) {
          this.tasks.set(cmd.id, { ...existing, ...cmd.updates });
        }
        break;
      case "delete":
        this.tasks.delete(cmd.id);
        break;
      case "snapshot":
        this.tasks.clear();
        for (const task of cmd.tasks) {
          this.tasks.set(task.id, task);
        }
        break;
    }
    this.notifyUpdate();
  }

  /**
   * 订阅变更
   */
  onUpdate(callback: UpdateCallback): () => void {
    this.listeners.add(callback);
    return () => this.listeners.delete(callback);
  }

  /**
   * 获取所有任务
   */
  getTasks(): Map<string, TaskRecord> {
    return this.tasks;
  }

  /**
   * 清空
   */
  clear(): void {
    this.tasks.clear();
    this.notifyUpdate();
  }

  private notifyUpdate(): void {
    for (const listener of this.listeners) {
      try {
        listener(new Map(this.tasks)); // 副本
      } catch (err) {
        console.error("[TimelineStore] listener error:", err);
      }
    }
  }
}
```

## 4. TimelineAdapter（核心转换逻辑）

`features/ai/lib/timeline-adapter.ts`

```typescript
import { TimelineStore } from "./timeline-store";
import { getNodeDisplayName, getNodeCategory } from "@/features/ai/graph/registry";
import type { TaskRecord } from "@/features/ai/types/timeline";

/**
 * Graph Chunk → TimelineCommand 转换
 */
export function adaptGraphChunk(
  nodeName: string,
  nodeOutput: Record<string, unknown>,
  onCommand: (cmd: ReturnType<typeof createTaskCmd> | ReturnType<typeof updateTaskCmd>) => void
): void {
  const displayName = getNodeDisplayName(nodeName);
  const category = getNodeCategory(nodeName);
  const executionId = `${nodeName}-${Date.now()}`;

  // Node 开始
  onCommand(createTaskCmd({
    id: executionId,
    title: displayName,
    category,
    status: "running",
    startTime: Date.now(),
  }));

  // Node 完成
  onCommand(updateTaskCmd(executionId, {
    status: "success",
    endTime: Date.now(),
    detail: extractDetail(nodeOutput),
  }));
}

function createTaskCmd(data: {
  id: string;
  title: string;
  category: TaskRecord["category"];
  status: TaskRecord["status"];
  startTime: number;
  detail?: string;
  metadata?: Record<string, unknown>;
}) {
  return {
    op: "create" as const,
    task: {
      id: data.id,
      parentId: null,
      title: data.title,
      status: data.status,
      detail: data.detail,
      startTime: data.startTime,
      category: data.category,
      metadata: data.metadata,
    },
  };
}

function updateTaskCmd(id: string, updates: Partial<TaskRecord>) {
  return { op: "update" as const, id, updates };
}

/**
 * 从 nodeOutput 提取 detail 信息
 */
function extractDetail(nodeOutput: Record<string, unknown>): string | undefined {
  // toolResults 中提取
  const toolResults = nodeOutput.toolResults;
  if (toolResults && typeof toolResults === "object") {
    for (const [toolName, result] of Object.entries(toolResults as Record<string, unknown>)) {
      if (result && typeof result === "object") {
        const r = result as Record<string, unknown>;

        // count 字段
        if (typeof r.count === "number") {
          return `找到 ${r.count} 条记录`;
        }

        // candidates 字段
        if (Array.isArray(r.candidates)) {
          return r.candidates
            .slice(0, 3)
            .map((c: unknown) => {
              if (typeof c === "object" && c !== null) {
                const obj = c as Record<string, unknown>;
                return obj.label ?? obj.name ?? JSON.stringify(c).slice(0, 20);
              }
              return String(c);
            })
            .join(" · ");
        }

        // resolvedEntities 字段
        if (r.resolvedEntities && typeof r.resolvedEntities === "object") {
          const entities = r.resolvedEntities as Record<string, unknown>;
          const resolved = Object.entries(entities)
            .filter(([, v]) => v !== null && v !== undefined)
            .map(([k]) => k);
          if (resolved.length > 0) {
            return `已确认：${resolved.join("、")}`;
          }
        }
      }
    }
  }

  // response 字段（最后生成的内容）
  if (typeof nodeOutput.response === "string" && nodeOutput.response.length > 0) {
    return nodeOutput.response.slice(0, 50) + (nodeOutput.response.length > 50 ? "..." : "");
  }

  return undefined;
}
```

## 5. TreeBuilder（扁平 → 嵌套）

`features/ai/ui/hooks/useTimelineTree.ts`

```typescript
import { useMemo } from "react";
import type { TaskRecord } from "@/features/ai/types/timeline";

export interface TreeNode {
  id: string;
  title: string;
  status: TaskRecord["status"];
  detail?: string;
  startTime: number;
  endTime?: number;
  duration?: number;
  category: TaskRecord["category"];
  children: TreeNode[];
  metadata?: Record<string, unknown>;
}

export function useTimelineTree(records: Map<string, TaskRecord>): TreeNode[] {
  return useMemo(() => {
    return buildTree(Array.from(records.values()));
  }, [records]);
}

function buildTree(records: TaskRecord[]): TreeNode[] {
  const nodes = new Map<string, TreeNode>();
  const roots: TreeNode[] = [];

  // 1. 创建所有节点
  for (const record of records) {
    nodes.set(record.id, {
      id: record.id,
      title: record.title,
      status: record.status,
      detail: record.detail,
      startTime: record.startTime,
      endTime: record.endTime,
      duration: record.endTime ? record.endTime - record.startTime : undefined,
      category: record.category,
      children: [],
      metadata: record.metadata,
    });
  }

  // 2. 建立父子关系
  for (const record of records) {
    const node = nodes.get(record.id)!;
    if (record.parentId && nodes.has(record.parentId)) {
      nodes.get(record.parentId)!.children.push(node);
    } else {
      roots.push(node);
    }
  }

  // 3. 按 startTime 排序
  const sortByTime = (arr: TreeNode[]) => {
    arr.sort((a, b) => a.startTime - b.startTime);
    arr.forEach((n) => sortByTime(n.children));
  };
  sortByTime(roots);

  return roots;
}
```

## 7. 前端 TimelineStore Hook

`features/ai/ui/hooks/useTimelineStore.ts`

```typescript
"use client";

import { useState, useEffect, useCallback, useRef } from "react";
import type { TaskRecord } from "@/features/ai/types/timeline";

export function useTimelineStore(initialTasks?: TaskRecord[]) {
  const [tasks, setTasks] = useState<Map<string, TaskRecord>>(new Map());
  const [collapsed, setCollapsed] = useState(false);
  const eventSourceRef = useRef<EventSource | null>(null);

  // 初始化
  useEffect(() => {
    if (initialTasks) {
      setTasks(new Map(initialTasks.map((t) => [t.id, t])));
    }
  }, [initialTasks]);

  // 订阅 SSE 事件
  const subscribe = useCallback((conversationId: string) => {
    const eventSource = new EventSource(`/api/ai/conversations/${conversationId}/timeline`);

    eventSource.onmessage = (event) => {
      try {
        const data = JSON.parse(event.data);

        if (data.type === "timeline_snapshot") {
          setTasks(new Map(data.tasks.map((t: TaskRecord) => [t.id, t])));
        }
      } catch (err) {
        console.error("[Timeline] SSE parse error:", err);
      }
    };

    eventSource.onerror = () => {
      console.error("[Timeline] SSE error");
      eventSource.close();
    };

    eventSourceRef.current = eventSource;

    return () => {
      eventSource.close();
    };
  }, []);

  // 切换折叠
  const toggle = useCallback(() => {
    setCollapsed((prev) => !prev);
  }, []);

  return {
    tasks,
    collapsed,
    toggle,
    subscribe,
  };
}
```

## 8. Timeline UI 组件

`features/ai/ui/AiThinkingTrace.tsx`

```tsx
"use client";

import { useTimelineTree, type TreeNode } from "@/features/ai/ui/hooks/useTimelineTree";
import { useTimelineStore } from "@/features/ai/ui/hooks/useTimelineStore";
import type { TaskRecord } from "@/features/ai/types/timeline";

interface AiThinkingTraceProps {
  conversationId: string;
  initialTasks?: TaskRecord[];
}

export function AiThinkingTrace({ conversationId, initialTasks }: AiThinkingTraceProps) {
  const { tasks, collapsed, toggle, subscribe } = useTimelineStore(initialTasks);
  const tree = useTimelineTree(tasks);

  // 订阅 SSE
  useEffect(() => {
    const unsubscribe = subscribe(conversationId);
    return unsubscribe;
  }, [conversationId, subscribe]);

  if (tree.length === 0) return null;

  return (
    <div className="timeline-container">
      <div className="timeline-header" onClick={toggle}>
        <span className="icon">{collapsed ? "▶" : "▼"}</span>
        <span className="title">AI 思考过程</span>
        <span className="count">({tree.length} 步)</span>
      </div>

      {!collapsed && (
        <div className="timeline-content">
          {tree.map((node) => (
            <TreeNodeComponent key={node.id} node={node} />
          ))}
        </div>
      )}
    </div>
  );
}

function TreeNodeComponent({ node, depth = 0 }: { node: TreeNode; depth?: number }) {
  const statusConfig = {
    pending: { icon: "○", className: "status-pending" },
    running: { icon: "●", className: "status-running" },
    success: { icon: "✓", className: "status-success" },
    warning: { icon: "⚠", className: "status-warning" },
    error: { icon: "✗", className: "status-error" },
  };

  const config = statusConfig[node.status];
  const categoryIcon = {
    reason: "🧠",
    tool: "🔧",
    workflow: "⚙",
    system: "⚡",
    human: "👤",
  }[node.category];

  return (
    <div className={`timeline-node ${config.className}`} style={{ marginLeft: depth * 16 }}>
      <span className="icon">{config.icon}</span>
      <span className="category">{categoryIcon}</span>
      <span className="title">{node.title}</span>
      {node.detail && <span className="detail">{node.detail}</span>}
      {node.duration && (
        <span className="duration">{node.duration < 1000 ? `${node.duration}ms` : `${(node.duration / 1000).toFixed(1)}s`}</span>
      )}

      {node.children.length > 0 && (
        <div className="children">
          {node.children.map((child) => (
            <TreeNodeComponent key={child.id} node={child} depth={depth + 1} />
          ))}
        </div>
      )}
    </div>
  );
}
```

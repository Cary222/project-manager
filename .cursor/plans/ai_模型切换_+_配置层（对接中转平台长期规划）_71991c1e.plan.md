#### 4. `features/ai/llm/providers/registry.ts`

```typescript
import { create AgnesModel } from "./agnes";
import { createDeepSeekModel } from "./deepseek";
import type { LLMProvider, ModelCatalogEntry } from "./types";

export const SYSTEM_PROVIDERS: LLMProvider[] = [
  {
    id: "agnes",
    name: "Agnes",
    description: "恒星研内部 AI 助手（默认）",
    baseURL: process.env.AGNES_API_URL ?? "https://apihub.agnes-ai.com/v1",
    defaultModel: "agnes-2.5-flash",
    isConfigured: Boolean(process.env.OPENAI_API_KEY),
  },
  {
    id: "deepseek",
    name: "DeepSeek",
    description: "DeepSeek Chat（需配置 API Key）",
    baseURL: "https://api.deepseek.com/v1",
    defaultModel: "deepseek-chat",
    isConfigured: Boolean(process.env.DEEPSEEK_API_KEY),
  },
];

export const MODEL_CATALOG: ModelCatalogEntry[] = [
  { publicName: "Agnes 2.5 Flash", providerId: "agnes", gatewayModel: "agnes-2.5-flash" },
  { publicName: "Agnes 2.0 Flash", providerId: "agnes", gatewayModel: "agnes-2.0-flash" },
  { publicName: "DeepSeek Chat", providerId: "deepseek", gatewayModel: "deepseek-chat" },
  { publicName: "DeepSeek Coder", providerId: "deepseek", gatewayModel: "deepseek-coder" },
];

export function createModel(providerId: string, modelOverride?: string) {
  const provider = SYSTEM_PROVIDERS.find(p => p.id === providerId);
  if (!provider) throw new Error(`Unknown provider: ${providerId}`);

  const modelName = modelOverride ?? provider.defaultModel;

  switch (providerId) {
    case "agnes":
      return create AgnesModel(modelName);
    case "deepseek":
      // DeepSeek API Key 从 env 读（第一版），后续接 DB
      return createDeepSeekModel(process.env.DEEPSEEK_API_KEY ?? "", modelName);
    default:
      throw new Error(`Unsupported provider: ${providerId}`);
  }
}

export function getAvailableModels(): ModelCatalogEntry[] {
  return MODEL_CATALOG.filter(entry => {
    const provider = SYSTEM_PROVIDERS.find(p => p.id === entry.providerId);
    return provider?.isConfigured;
  });
}
```

### Stage 3：前端模型选择下拉

#### 8. `features/ai/llm/model-selector.tsx`

```tsx
"use client";
import { useState, useEffect, useCallback } from "react";
import type { ModelCatalogEntry } from "./types";

interface ModelSelectorProps {
  value: string;           // 当前选中的 gatewayModel
  onChange: (model: string) => void;
}

export function ModelSelector({ value, onChange }: ModelSelectorProps) {
  const [models, setModels] = useState<ModelCatalogEntry[]>([]);
  const [open, setOpen] = useState(false);

  useEffect(() => {
    fetch("/api/ai/models")
      .then(r => r.json())
      .then(d => setModels(d.data ?? []));
  }, []);

  const current = models.find(m => m.gatewayModel === value);

  return (
    <div className="relative">
      <button
        onClick={() => setOpen(!open)}
        className="flex items-center gap-1.5 rounded-lg border border-ink-200 bg-white px-2 py-1 text-xs font-medium text-ink-700 hover:bg-ink-50"
      >
        <span>{current?.publicName ?? "选择模型"}</span>
        <ChevronDown className="h-3 w-3" />
      </button>

      {open && (
        <div className="absolute right-0 top-full z-50 mt-1 w-48 rounded-xl border border-ink-200 bg-white py-1 shadow-base">
          {models.map(model => (
            <button
              key={model.gatewayModel}
              onClick={() => { onChange(model.gatewayModel); setOpen(false); }}
              className={`flex w-full items-center justify-between px-3 py-2 text-sm hover:bg-ink-50 ${
                model.gatewayModel === value ? "bg-brand-50 text-brand-700" : "text-ink-700"
              }`}
            >
              <span>{model.publicName}</span>
              {model.gatewayModel === value && <IconCheck className="h-4 w-4" />}
            </button>
          ))}
        </div>
      )}
    </div>
  );
}
```
